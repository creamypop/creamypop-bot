# Callback for Payment Button -> Sends QR Code Photo
@bot.callback_query_handler(func=lambda call: call.data == "show_payment")
def send_qr_code(call):
    bot.answer_callback_query(call.id)
    payment_instruction = (
        "💳 **Payment Details:**\n\n"
        "1. Scan the QR code below to pay **₹129**.\n"
        "2. After payment, **send the payment screenshot here** as an image.\n"
        "3. Once verified, you will receive the private channel link automatically!"
    )
    
    try:
        with open("qr_code.jpg", "rb") as qr:
            bot.send_photo(call.message.chat.id, photo=qr, caption=payment_instruction)

        bot.send_message(call.message.chat.id, payment_instruction + "\n\n(Please send screenshot after payment)")

# Step 5 & 6: User sends Payment Screenshot
@bot.message_handler(content_types=['photo'])
def handle_screenshot(message):
    user_id = message.from_user.id
    user_name = message.from_user.first_name or "User"
    photo_file_id = message.photo[-1].file_id

    # Reply to User
    bot.reply_to(message, "⏳ Please wait for approval so I can manually verify your payment.")

    # Forward to Admin with Approve / Reject buttons
    admin_markup = InlineKeyboardMarkup()
    approve_btn = InlineKeyboardButton("✅ Approve", callback_data=f"approve_{user_id}")
    reject_btn = InlineKeyboardButton("❌ Reject", callback_data=f"reject_{user_id}")
    admin_markup.add(approve_btn, reject_btn)

    bot.send_photo(
        ADMIN_ID, 
        photo=photo_file_id, 
        caption=f"📩 **New Payment Screenshot!**\n\nUser: {user_name}\nID: `{user_id}`",
        reply_markup=admin_markup
    )

# Step 7: Admin Approval / Rejection Callback
@bot.callback_query_handler(func=lambda call: call.data.startswith("approve_") or call.data.startswith("reject_"))
def process_admin_decision(call):
    action, target_user_id = call.data.split("_")
    target_user_id = int(target_user_id)

    if call.from_user.id != ADMIN_ID:
        bot.answer_callback_query(call.id, "Unauthorized!", show_alert=True)
        return

    if action == "approve":
        # Send channel link to user
        bot.send_message(
            target_user_id,
            f"🎉 **Payment Verified!**\n\nHere is your exclusive access link:\n{CHANNEL_LINK}"
        )
        bot.edit_message_caption(
            chat_id=ADMIN_ID,
            message_id=call.message.message_id,
            caption=call.message.caption + "\n\n✅ **APPROVED**"
        )
        bot.answer_callback_query(call.id, "Approved and link sent.")
    
    elif action == "reject":
        bot.send_message(
            target_user_id,
            "❌ **Payment Verification Failed.**\nPlease check your screenshot or contact support."
        )
        bot.edit_message_caption(
            chat_id=ADMIN_ID,
            message_id=call.message.message_id,
            caption=call.message.caption + "\n\n❌ **REJECTED**"
        )
        bot.answer_callback_query(call.id, "Rejected.")
